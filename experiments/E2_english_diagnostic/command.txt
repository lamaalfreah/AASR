# ---------------------------------------------------------------------------
# Budget calibration (8 problems each) — why --max-new-floor exists
#
# E0's budget formula is max_new_tokens = 256 + alpha * 768, calibrated on
# Arabic answers averaging ~7 tokens. Unchanged on MATH500 it truncated most
# generations before the FINAL line, so accuracy measured token budget rather
# than reasoning. These three runs found the point of diminishing returns.
#
#   floor 256  (E0 default) -> alpha 0: 25.0% accuracy, 62.5% truncated
#   floor 2048              -> alpha 0: 62.5% accuracy,  0.0% truncated
#   floor 4096              -> alpha 0: 62.5% accuracy,  0.0% truncated
#                              alpha 1 gained ~500 tokens, no accuracy change
#
# 4096 was adopted for the main run and applied equally to every alpha.
# ---------------------------------------------------------------------------
modal run training/ASAR/modal_runner_e2_english.py::run_eval \
  --alphas 0,0.25,1 --max-samples 8 --seed 42

modal run training/ASAR/modal_runner_e2_english.py::run_eval \
  --alphas 0,0.25,1 --max-samples 8 --seed 42 --max-new-floor 2048

modal run training/ASAR/modal_runner_e2_english.py::run_eval \
  --alphas 0,0.25,1 --max-samples 8 --seed 42 --max-new-floor 4096
